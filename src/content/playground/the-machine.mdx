  // layer colors — muted tones that distinguish layers without clashing
  var lightColors = [
    { fill: '#f5e6d8', stroke: '#c49470', color: '#3a2a1a' },
    { fill: '#f0eacc', stroke: '#b8a44a', color: '#3a3420' },

    { fill: '#e4d8e8', stroke: '#7a5a8a', color: '#2a1a3a' }
  ];
  var darkColors = [
    { fill: '#3a2820', stroke: '#c49470', color: '#f0e0d0' },
    { fill: '#32301a', stroke: '#b8a44a', color: '#f0eacc' },
    { fill: '#1a3020', stroke: '#5a8a5a', color: '#d8e8d8' },

  function buildSyntax(system) {
    var isDark = document.documentElement.dataset.theme === 'dark';
    var cl = isDark ? darkColors : lightColors;
    var lines = ['flowchart TD'];

    // color classes
    cl.forEach(function(c, i) {
      lines.push('  classDef layer' + i + ' fill:' + c.fill + ',stroke:' + c.stroke + ',color:' + c.color);
    });

    // subgraphs + nodes
    system.layers.forEach(function(layer, li) {
      lines.push('  subgraph sg' + li + '[\\"' + layer.name + '\\"]');
      layer.nodes.forEach(function(n) {
        lines.push('    ' + n.id + n.s[0] + n.label + n.s[1]);
      });
      lines.push('  end');
    });

    // edges
    system.edges.forEach(function(e) {
      lines.push('  ' + e.f + ' -->|' + e.l + '| ' + e.t);
    });

    // class assignments per layer
    system.layers.forEach(function(layer, li) {
      var ci = li % cl.length;
      var ids = layer.nodes.map(function(n) { return n.id; }).join(',');
      lines.push('  class ' + ids + ' layer' + ci);
    });

    return lines.join('\\n');
  }

  function setupAnimation(wrapper, order) {
    var svg = wrapper.querySelector('svg');
    if (!svg) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      wrapper.classList.add('mermaid--visible');
      return;
    }

    var edgePaths = svg.querySelectorAll('.edgePaths > path[data-id]');
    var edgeLabels = svg.querySelectorAll('.edgeLabels > .edgeLabel');
    var delay = 0;

    svg.querySelectorAll('.node').forEach(function(n) { n.style.animationDelay = '99s'; });
    edgePaths.forEach(function(e) {
      var len = e.getTotalLength();
      e.style.strokeDasharray = len;
      e.style.strokeDashoffset = len;
      e.style.animationDelay = '99s';
    });
    edgeLabels.forEach(function(l) { l.style.animationDelay = '99s'; });

    order.forEach(function(name) {
      var node = svg.querySelector('.node[id*="-flowchart-' + name + '-"]');
      if (node) {
        node.style.animationDelay = delay + 's';
        delay += 0.3;
      }
      edgePaths.forEach(function(edge, i) {
        var parts = (edge.getAttribute('data-id') || '').split('_');
        if (parts.length >= 3 && parts[1] === name) {
          edge.style.animationDelay = delay + 's';
          delay += 0.24;
          if (edgeLabels[i]) {
            edgeLabels[i].style.animationDelay = (delay - 0.12) + 's';
          }
        }
      });
    });

    var totalDuration = (delay + 0.8) * 1000;
    var observer = new IntersectionObserver(function(entries) {
      entries.forEach(function(entry) {
        if (entry.isIntersecting) {
          wrapper.classList.add('mermaid--visible');
          observer.unobserve(wrapper);
          setTimeout(function() {
            edgePaths.forEach(function(e) {
              e.style.strokeDasharray = '';
              e.style.strokeDashoffset = '';
            });
          }, totalDuration);
        }
      });
    }, { threshold: 0.2 });
    observer.observe(wrapper);
  }

  // --- state ---
  var container = document.getElementById('machine-diagram');
  var wrapper = document.getElementById('machine-wrapper');
  var titleEl = document.getElementById('machine-title');
  var descEl = document.getElementById('machine-description');
  var counterEl = document.getElementById('machine-counter');
  var shuffleBtn = document.getElementById('machine-shuffle');
  if (!container || !wrapper) return;

  var shown = [];
  var mermaidReady = false;

  function pickSystem() {
    if (shown.length >= S.length) shown = [];
    var available = [];
    for (var i = 0; i < S.length; i++) {
      if (shown.indexOf(i) === -1) available.push(i);
    }
    var idx = available[Math.floor(Math.random() * available.length)];
    shown.push(idx);
    return { system: S[idx], num: shown.length };
  }

  function render(immediate) {
    var pick = pickSystem();
    var sys = pick.system;

    titleEl.textContent = sys.title;
    descEl.textContent = sys.desc;
    counterEl.textContent = pick.num + ' of ' + S.length;

    if (!mermaidReady) return;

    wrapper.classList.remove('mermaid--visible');
    var syntax = buildSyntax(sys);
    var order = [];
    sys.layers.forEach(function(layer) {
      layer.nodes.forEach(function(n) { order.push(n.id); });
    });

    window.mermaid.render('machine' + Date.now(), syntax).then(function(result) {
      container.innerHTML = result.svg;
      setupAnimation(wrapper, order);
      if (immediate) {
        wrapper.classList.add('mermaid--visible');
      }
    });
  }

  // load mermaid
  var s = document.createElement('script');
  s.src = 'https://cdn.jsdelivr.net/npm/mermaid@11/dist/mermaid.min.js';
  s.onload = function() {
    var isDark = document.documentElement.dataset.theme === 'dark';
    window.mermaid.initialize({
      startOnLoad: false,
      theme: isDark ? 'dark' : 'default',
      themeVariables: isDark
        ? { primaryColor: '#2d4a3e', primaryTextColor: '#e8e0d4', lineColor: '#5a7a6a' }
        : { primaryColor: '#e8e0d4', primaryTextColor: '#2c2c2c', lineColor: '#5a7a6a' }
    });
    mermaidReady = true;
    render(false);
  };
  document.head.appendChild(s);

  shuffleBtn.addEventListener('click', function() {
    render(true);
  });
})();
