<h2 class="howto-heading">How it works</h2>

<p>Each system is a data object - nodes grouped into layers, edges defining connections. No hand-drawn SVG, no image files. The diagram is built from structure.</p>

<pre class="machine-code"><code>{`// each system defines its architecture as data
{
  title: 'How a Web Page Loads',
  desc: 'Your browser checks its cache, resolves DNS...',

  // nodes are organized into semantic layers
  // mermaid renders them as subgraphs
  layers: [
    { name: 'Client', nodes: [
      { id: 'BR',  label: 'Browser' },
      { id: 'CA',  label: 'Browser Cache' }
    ]},
    { name: 'Resolution', nodes: [
      { id: 'DNS', label: 'DNS Resolver' },
      { id: 'TLS', label: 'TLS Handshake' }
    ]},
    // ...more layers
  ],

  // edges define data flow between nodes
  // the animation follows these in order
  edges: [
    { f: 'BR',  t: 'CA',  l: 'check cache' },
    { f: 'CA',  t: 'DNS', l: 'cache miss' },
    { f: 'DNS', t: 'TLS', l: 'IP address' },
    // ...more edges
  ]
}`}</code></pre>

<p>At render time, this data compiles into <a href="https://mermaid.js.org">Mermaid</a> flowchart syntax - subgraphs for layers, styled nodes, labeled edges. Mermaid handles the layout algorithm and SVG generation.</p>

<pre class="machine-code"><code>{`// compiled output sent to mermaid.render()
flowchart TD
  subgraph sg0["Client"]
    BR(["Browser"])
    CA(["Browser Cache"])
  end
  subgraph sg1["Resolution"]
    DNS[/"DNS Resolver"/]
    TLS[/"TLS Handshake"/]
  end
  BR -->|check cache| CA
  CA -->|cache miss| DNS
  DNS -->|IP address| TLS`}</code></pre>

<p>The animation system walks the SVG after mermaid renders it. Nodes start invisible, then fade in one by one. Edges draw themselves using <code>stroke-dashoffset</code> transitions. Arrowheads are hidden until their edge finishes drawing. The whole sequence chains through <code>setTimeout</code> callbacks - each step triggers the next.</p>

<p>Visual style is randomized per diagram. Some render as clean schematics, others get mermaid's <code>handDrawn</code> mode which uses <a href="https://roughjs.com">rough.js</a> under the hood for a sketched aesthetic. The style is injected as a mermaid directive before compilation.</p>

<pre class="machine-code"><code>{`// 50/50 chance of hand-drawn per diagram
var style = Math.random() > 0.5 ? 'handDrawn' : 'classic';

// injected as a mermaid config directive
var directive = style === 'handDrawn'
  ? '%%{init: {"look": "handDrawn"}}%%\\n'
  : '';

mermaid.render(id, directive + syntax);`}</code></pre>

<p>10 systems. 2 visual styles. Animated on scroll, instant on shuffle. All generated from ~500 lines of data.</p>

  function buildSyntax(system) {
    var isDark = document.documentElement.dataset.theme === 'dark';
    var cl = isDark ? darkColors : lightColors;
    var lines = ['flowchart TD'];

    // color classes
    cl.forEach(function(c, i) {
      lines.push('  classDef layer' + i + ' fill:' + c.fill + ',stroke:' + c.stroke + ',color:' + c.color);
    });

    // subgraphs + nodes
    system.layers.forEach(function(layer, li) {
      lines.push('  subgraph sg' + li + '[\\"' + layer.name + '\\"]');
      layer.nodes.forEach(function(n) {
        lines.push('    ' + n.id + n.s[0] + n.label + n.s[1]);
      });
      lines.push('  end');
    });

    // edges
    system.edges.forEach(function(e) {
      lines.push('  ' + e.f + ' -->|' + e.l + '| ' + e.t);
    });

    // class assignments per layer
    system.layers.forEach(function(layer, li) {
      var ci = li % cl.length;
      var ids = layer.nodes.map(function(n) { return n.id; }).join(',');
      lines.push('  class ' + ids + ' layer' + ci);
    });

    return lines.join('\\n');
  }

  function setupAnimation(wrapper, order, sys, immediate) {
    var svg = wrapper.querySelector('svg');
    if (!svg) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      wrapper.classList.add('mermaid--visible');
      return;
    }

    // same sequential approach as Base.astro: node fades in → edges draw →
    // labels appear → next node. JS-driven transitions, no CSS keyframes.

    var edges = svg.querySelectorAll('.edgePaths > path[data-id]');
    var edgeLabels = svg.querySelectorAll('.edgeLabels > .edgeLabel');

    // hide all nodes and edge labels
    svg.querySelectorAll('.node').forEach(function(n) { n.style.opacity = '0'; });
    edgeLabels.forEach(function(l) { l.style.opacity = '0'; });

    // prep edges: dash-hide strokes, save and remove arrowhead markers
    edges.forEach(function(e) {
      var len = e.getTotalLength();
      e.style.strokeDasharray = len;
      e.style.strokeDashoffset = len;
      e._savedMarker = e.getAttribute('marker-end');
      e.removeAttribute('marker-end');
    });

    // build sequential step list from node order
    var steps = [];
    order.forEach(function(name) {
      var node = svg.querySelector('.node[id*="-flowchart-' + name + '-"]');
      if (node) steps.push({ type: 'node', el: node });
      var nodeEdges = [];
      edges.forEach(function(edge, i) {
        var parts = (edge.getAttribute('data-id') || '').split('_');
        if (parts.length >= 3 && parts[1] === name) {
          nodeEdges.push({ el: edge, label: edgeLabels[i] || null });
        }
      });
      if (nodeEdges.length) steps.push({ type: 'edges', items: nodeEdges });
    });

    function runStep(i) {
      if (i >= steps.length) {
        // cleanup inline styles so diagram looks clean after animation
        edges.forEach(function(e) {
          e.style.strokeDasharray = '';
          e.style.strokeDashoffset = '';
          e.style.transition = '';
        });
        return;
      }
      var step = steps[i];
      if (step.type === 'node') {
        step.el.style.transition = 'opacity 0.35s ease-out';
        step.el.style.opacity = '1';
        // scale pop on inner shape — skip polygons (diamonds rotate weirdly)
        var shape = step.el.querySelector('rect, circle');
        if (shape) {
          shape.style.transformOrigin = 'center center';
          shape.style.transformBox = 'fill-box';
          shape.style.transform = 'scale(1.18)';
          shape.style.transition = 'transform 0.5s cubic-bezier(0.34, 1.56, 0.64, 1)';
          setTimeout(function() { shape.style.transform = 'scale(1)'; }, 50);
        }
        setTimeout(function() { runStep(i + 1); }, 450);
      } else if (step.type === 'edges') {
        step.items.forEach(function(e) {
          e.el.style.transition = 'stroke-dashoffset 0.7s cubic-bezier(0.4, 0, 0.2, 1)';
          e.el.style.strokeDashoffset = '0';
        });
        setTimeout(function() {
          // restore arrowheads and fade in labels
          step.items.forEach(function(e) {
            if (e.el._savedMarker) e.el.setAttribute('marker-end', e.el._savedMarker);
            if (e.label) {
              e.label.style.transition = 'opacity 0.3s ease-out';
              e.label.style.opacity = '1';
            }
          });
          runStep(i + 1);
        }, 650);
      }
    }

    if (immediate) {
      // shuffle — user is already looking at the diagram
      wrapper.classList.add('mermaid--visible');
      runStep(0);
    } else {
      // initial load — wait until diagram scrolls into view
      var observer = new IntersectionObserver(function(entries) {
        entries.forEach(function(entry) {
          if (entry.isIntersecting) {
            wrapper.classList.add('mermaid--visible');
            observer.unobserve(wrapper);
            runStep(0);
          }
        });
      }, { threshold: 0.3 });
      observer.observe(wrapper);
    }
  }
