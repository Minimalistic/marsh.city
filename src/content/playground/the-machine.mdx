<h2 class="howto-heading">How it works</h2>

<p>Each system is a data object - nodes grouped into layers, edges defining connections. No hand-drawn SVG, no image files. The diagram is built from structure.</p>

<pre class="machine-code"><code>{`// each system defines its architecture as data
{
  title: 'How a Web Page Loads',
  desc: 'Your browser checks its cache, resolves DNS...',

  // nodes are organized into semantic layers
  // mermaid renders them as subgraphs
  layers: [
    { name: 'Client', nodes: [
      { id: 'BR',  label: 'Browser' },
      { id: 'CA',  label: 'Browser Cache' }
    ]},
    { name: 'Resolution', nodes: [
      { id: 'DNS', label: 'DNS Resolver' },
      { id: 'TLS', label: 'TLS Handshake' }
    ]},
    // ...more layers
  ],

  // edges define data flow between nodes
  // the animation follows these in order
  edges: [
    { f: 'BR',  t: 'CA',  l: 'check cache' },
    { f: 'CA',  t: 'DNS', l: 'cache miss' },
    { f: 'DNS', t: 'TLS', l: 'IP address' },
    // ...more edges
  ]
}`}</code></pre>

<p>At render time, this data compiles into <a href="https://mermaid.js.org">Mermaid</a> flowchart syntax - subgraphs for layers, styled nodes, labeled edges. Mermaid handles the layout algorithm and SVG generation.</p>

<pre class="machine-code"><code>{`// compiled output sent to mermaid.render()
flowchart TD
  subgraph sg0["Client"]
    BR(["Browser"])
    CA(["Browser Cache"])
  end
  subgraph sg1["Resolution"]
    DNS[/"DNS Resolver"/]
    TLS[/"TLS Handshake"/]
  end
  BR -->|check cache| CA
  CA -->|cache miss| DNS
  DNS -->|IP address| TLS`}</code></pre>

<p>The animation system walks the SVG after mermaid renders it. Nodes start invisible, then fade in one by one. Edges draw themselves using <code>stroke-dashoffset</code> transitions. Arrowheads are hidden until their edge finishes drawing. The whole sequence chains through <code>setTimeout</code> callbacks - each step triggers the next.</p>

<p>Visual style is randomized per diagram. Some render as clean schematics, others get mermaid's <code>handDrawn</code> mode which uses <a href="https://roughjs.com">rough.js</a> under the hood for a sketched aesthetic. The style is injected as a mermaid directive before compilation.</p>

<pre class="machine-code"><code>{`// 50/50 chance of hand-drawn per diagram
var style = Math.random() > 0.5 ? 'handDrawn' : 'classic';

// re-initialize mermaid with the chosen look
// handDrawn activates rough.js for a sketchy aesthetic
mermaid.initialize({ startOnLoad: false, look: style });
mermaid.render(id, syntax);`}</code></pre>

<p>10 systems. 2 visual styles. Animated on scroll, instant on shuffle. All generated from ~500 lines of data.</p>

  function buildSyntax(system) {
    var isDark = document.documentElement.dataset.theme === 'dark';
    var cl = isDark ? darkColors : lightColors;
    var lines = ['flowchart TD'];

    // color classes
    cl.forEach(function(c, i) {
      lines.push('  classDef layer' + i + ' fill:' + c.fill + ',stroke:' + c.stroke + ',color:' + c.color);
    });

    // subgraphs + nodes
    system.layers.forEach(function(layer, li) {
      lines.push('  subgraph sg' + li + '[\\"' + layer.name + '\\"]');
      layer.nodes.forEach(function(n) {
        lines.push('    ' + n.id + n.s[0] + n.label + n.s[1]);
      });
      lines.push('  end');
    });

    // edges
    system.edges.forEach(function(e) {
      lines.push('  ' + e.f + ' -->|' + e.l + '| ' + e.t);
    });

    // class assignments per layer
    system.layers.forEach(function(layer, li) {
      var ci = li % cl.length;
      var ids = layer.nodes.map(function(n) { return n.id; }).join(',');
      lines.push('  class ' + ids + ' layer' + ci);
    });

    return lines.join('\\n');
  }

  function setupAnimation(wrapper, order, sys, immediate) {
    var svg = wrapper.querySelector('svg');
    if (!svg) return;
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      wrapper.classList.add('mermaid--visible');
      return;
    }

    // same sequential approach as Base.astro: node fades in → edges draw →
    // labels appear → next node. JS-driven transitions, no CSS keyframes.

    // select only actual edge paths (not arrowhead markers)
    var allEdgePaths = svg.querySelectorAll('.edgePaths path');
    var edges = [];
    allEdgePaths.forEach(function(p) {
      if (p.classList.contains('arrowMarkerPath')) return;
      edges.push(p);
    });
    var edgeLabels = svg.querySelectorAll('.edgeLabels > .edgeLabel');

    // hide all nodes and edge labels
    // use both selectors — .node works in schematic, [id*=flowchart] in hand-drawn
    var allNodes = svg.querySelectorAll('.node');
    if (allNodes.length === 0) allNodes = svg.querySelectorAll('g[id*="flowchart-"]');
    allNodes.forEach(function(n) { n.style.opacity = '0'; });
    edgeLabels.forEach(function(l) { l.style.opacity = '0'; });

    // prep edges: thicken strokes for visible animation, dash-hide, save arrowheads
    edges.forEach(function(e) {
      e.style.strokeWidth = '2px';
      var len = e.getTotalLength();
      e.style.strokeDasharray = len;
      e.style.strokeDashoffset = len;
      e._savedMarker = e.getAttribute('marker-end');
      e.removeAttribute('marker-end');
    });

    // build sequential step list from node order
    var steps = [];
    order.forEach(function(name) {
      // match node by ID pattern — works in both schematic (.node class) and hand-drawn
      var node = svg.querySelector('[id*="-flowchart-' + name + '-"]');
      if (node) steps.push({ type: 'node', el: node });
      var nodeEdges = [];
      edges.forEach(function(edge, i) {
        var did = edge.getAttribute('data-id') || '';
        // data-id format: L_FROM_TO_0 — match source node after first underscore
        var match = did.match(/^L_([^_]+)_/);
        if (match && match[1] === name) {
          nodeEdges.push({ el: edge, label: edgeLabels[i] || null });
        }
      });
      if (nodeEdges.length) steps.push({ type: 'edges', items: nodeEdges });
    });

    // particle emitter — glowing leading dot + trail particles along edge path
    function emitParticles(pathEl, svgEl, duration) {
      var totalLen = pathEl.getTotalLength();
      var ns = 'http://www.w3.org/2000/svg';
      // match the edge's stroke color for all particles
      var edgeColor = getComputedStyle(pathEl).stroke || '#00ffcc';

      // leading glow dot — travels the full path length
      var lead = document.createElementNS(ns, 'circle');
      lead.setAttribute('r', '4');
      lead.setAttribute('fill', edgeColor);
      lead.setAttribute('opacity', '1');
      lead.style.filter = 'drop-shadow(0 0 6px ' + edgeColor + ') drop-shadow(0 0 12px ' + edgeColor + ')';
      svgEl.appendChild(lead);

      var leadStart = performance.now();
      function moveLead(now) {
        var t = Math.min((now - leadStart) / duration, 1);
        // match the edge draw easing: cubic-bezier(0.4, 0, 0.2, 1)
        var ease = t < 0.5
          ? 4 * t * t * t
          : 1 - Math.pow(-2 * t + 2, 3) / 2;
        var pt = pathEl.getPointAtLength(ease * totalLen);
        lead.setAttribute('cx', pt.x);
        lead.setAttribute('cy', pt.y);
        if (t < 1) {
          requestAnimationFrame(moveLead);
        } else {
          // fade out the leading dot
          var fadeStart = performance.now();
          function fadeLead(now2) {
            var ft = Math.min((now2 - fadeStart) / 200, 1);
            lead.setAttribute('opacity', String(1 - ft));
            if (ft < 1) requestAnimationFrame(fadeLead);
            else lead.remove();
          }
          requestAnimationFrame(fadeLead);
        }
      }
      requestAnimationFrame(moveLead);

      // trail particles — shed from the leading dot at intervals
      var particleCount = 10;
      var interval = duration / particleCount;
      for (var p = 0; p < particleCount; p++) {
        (function(idx) {
          setTimeout(function() {
            var progress = idx / particleCount;
            var pt = pathEl.getPointAtLength(progress * totalLen);
            var circle = document.createElementNS(ns, 'circle');
            circle.setAttribute('cx', pt.x);
            circle.setAttribute('cy', pt.y);
            circle.setAttribute('r', '2');
            circle.setAttribute('fill', edgeColor);
            circle.setAttribute('opacity', '0.8');
            circle.style.filter = 'drop-shadow(0 0 3px ' + edgeColor + ')';
            svgEl.appendChild(circle);

            var angle = Math.random() * Math.PI * 2;
            var drift = 6 + Math.random() * 10;
            var dx = Math.cos(angle) * drift;
            var dy = Math.sin(angle) * drift;
            var startTime = performance.now();
            var fadeMs = 350 + Math.random() * 250;

            function tick(now) {
              var t = Math.min((now - startTime) / fadeMs, 1);
              var ease = 1 - (1 - t) * (1 - t);
              circle.setAttribute('cx', pt.x + dx * ease);
              circle.setAttribute('cy', pt.y + dy * ease);
              circle.setAttribute('opacity', String(0.8 * (1 - ease)));
              circle.setAttribute('r', String(2 * (1 - ease * 0.5)));
              if (t < 1) {
                requestAnimationFrame(tick);
              } else {
                circle.remove();
              }
            }
            requestAnimationFrame(tick);
          }, idx * interval);
        })(p);
      }
    }

    function runStep(i) {
      if (i >= steps.length) {
        // cleanup inline styles so diagram looks clean after animation
        edges.forEach(function(e) {
          e.style.strokeDasharray = '';
          e.style.strokeDashoffset = '';
          e.style.transition = '';
        });
        return;
      }
      var step = steps[i];
      if (step.type === 'node') {
        step.el.style.transition = 'opacity 0.35s ease-out';
        step.el.style.opacity = '1';
        // scale pop on inner shape — skip polygons (diamonds rotate weirdly)
        var shape = step.el.querySelector('rect, circle');
        if (shape) {
          shape.style.transformOrigin = 'center center';
          shape.style.transformBox = 'fill-box';
          shape.style.transform = 'scale(1.18)';
          shape.style.transition = 'transform 0.5s cubic-bezier(0.34, 1.56, 0.64, 1)';
          setTimeout(function() { shape.style.transform = 'scale(1)'; }, 50);
        }
        setTimeout(function() { runStep(i + 1); }, 450);
      } else if (step.type === 'edges') {
        step.items.forEach(function(e) {
          e.el.style.transition = 'stroke-dashoffset 0.7s cubic-bezier(0.4, 0, 0.2, 1)';
          e.el.style.strokeDashoffset = '0';
          // spawn particles along the leading edge
          emitParticles(e.el, svg, 700);
        });
        setTimeout(function() {
          // restore arrowheads and fade in labels
          step.items.forEach(function(e) {
            if (e.el._savedMarker) e.el.setAttribute('marker-end', e.el._savedMarker);
            if (e.label) {
              e.label.style.transition = 'opacity 0.3s ease-out';
              e.label.style.opacity = '1';
            }
          });
          runStep(i + 1);
        }, 650);
      }
    }

    if (immediate) {
      // shuffle — user is already looking at the diagram
      wrapper.classList.add('mermaid--visible');
      runStep(0);
    } else {
      // initial load — wait until diagram scrolls into view
      var observer = new IntersectionObserver(function(entries) {
        entries.forEach(function(entry) {
          if (entry.isIntersecting) {
            wrapper.classList.add('mermaid--visible');
            observer.unobserve(wrapper);
            runStep(0);
          }
        });
      }, { threshold: 0.3 });
      observer.observe(wrapper);
    }
  }

  }

    window.mermaid.initialize({
      startOnLoad: false,
      look: style,
      theme: 'dark',
      themeVariables: {
        primaryColor: '#0a1a1a',
        primaryTextColor: '#00ffcc',
        lineColor: '#00ffcc',
        secondaryColor: '#1a0a1a',
        tertiaryColor: '#0a0a1a',
        primaryBorderColor: '#00ffcc',
        background: '#0a0e14'
      }
    });

    var syntax = buildSyntax(sys);
    var order = [];
    sys.layers.forEach(function(layer) {
      layer.nodes.forEach(function(n) { order.push(n.id); });
    });

    window.mermaid.render('machine' + Date.now(), syntax).then(function(result) {
      container.innerHTML = result.svg;
      // force transparent background on the SVG mermaid generates
      var svgEl = container.querySelector('svg');
      if (svgEl) {
        svgEl.style.background = 'transparent';
        var bgRect = svgEl.querySelector('rect.er');
        if (bgRect) bgRect.setAttribute('fill', 'transparent');

      // strip diagonal hachure fill lines from node shapes in both modes.
      // rough.js (hand-drawn) and mermaid's default renderer can both produce
      // fill patterns — keep only the outline path (last in each shape group)
      if (svgEl) {
        var nodes = svgEl.querySelectorAll('.node, g[id*="flowchart-"]');
        nodes.forEach(function(node) {
          var innerGroups = node.querySelectorAll('g');

            if (paths.length > 1) {
              for (var pi = 0; pi < paths.length - 1; pi++) {
                if (paths[pi].getAttribute('fill') === 'none' ||
                    !paths[pi].getAttribute('fill')) {
                  paths[pi].style.display = 'none';
                }
              }
            }
          });
        });
      }

      setupAnimation(wrapper, order, sys, immediate);
    });
  }

  // load mermaid
  var s = document.createElement('script');
  s.src = 'https://cdn.jsdelivr.net/npm/mermaid@11/dist/mermaid.min.js';
  s.onload = function() {
    window.mermaid.initialize({ startOnLoad: false });
    mermaidReady = true;
